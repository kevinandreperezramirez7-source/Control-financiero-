<script>
const SUPABASE_URL = "https://ifsphpzvjzoktibazkmi.supabase.co";
const SUPABASE_KEY = "sb_publishable_7KraiaSY-0cQ7OoMFQruMA_6-73sHeh";

const $ = id => document.getElementById(id);
let sb = null, usuarioActual = null, meses = [], movimientos = [];

const configurado = /^https:\/\/.+\.supabase\.co\/?$/.test(SUPABASE_URL.trim()) &&
                    SUPABASE_KEY.trim() && !SUPABASE_KEY.includes("PON_AQUI");

if (configurado) {
  sb = supabase.createClient(SUPABASE_URL.trim(), SUPABASE_KEY.trim());
} else {
  const e = $("configError");
  e.classList.remove("hidden");
  e.innerHTML = "⚠️ Falta configurar Supabase. Revisa la URL y la clave en el código.";
}

function dinero(n){
  return new Intl.NumberFormat("es-GT",{style:"currency",currency:"GTQ"}).format(n);
}

function hoyLocal(){
  const d = new Date();
  return d.getFullYear() + "-" + String(d.getMonth()+1).padStart(2,"0") + "-" + String(d.getDate()).padStart(2,"0");
}

function mostrarMensaje(texto,color){
  $("loginMessage").textContent = texto;
  $("loginMessage").style.color = color;
}

function listo(){
  if (!sb) { mostrarMensaje("Primero configura la URL y la clave de Supabase en el código.","red"); return false; }
  return true;
}

async function registrarse(){
  if(!listo()) return;
  const email = $("email").value.trim(), password = $("password").value;
  if(!email || !password) return mostrarMensaje("Escribe tu correo y contraseña.","red");
  if(password.length < 6) return mostrarMensaje("La contraseña debe tener mínimo 6 caracteres.","red");
  const {data,error} = await sb.auth.signUp({email,password});
  if(error) return mostrarMensaje(error.message,"red");
  if(data.session){ usuarioActual = data.user; return mostrarAplicacion(); }
  mostrarMensaje("Cuenta creada. Revisa tu correo para confirmarla y luego inicia sesión.","green");
}

async function iniciarSesion(){
  if(!listo()) return;
  const email = $("email").value.trim(), password = $("password").value;
  if(!email || !password) return mostrarMensaje("Completa todos los campos.","red");
  const {data,error} = await sb.auth.signInWithPassword({email,password});
  if(error){
    const msg = /confirm/i.test(error.message) ? "Debes confirmar tu correo antes de entrar." : "Correo o contraseña incorrectos.";
    return mostrarMensaje(msg,"red");
  }
  usuarioActual = data.user;
  mostrarAplicacion();
}

["email","password"].forEach(id => $(id).addEventListener("keydown", e => { if(e.key==="Enter") iniciarSesion(); }));

async function mostrarAplicacion(){
  $("loginScreen").classList.add("hidden");
  $("app").classList.remove("hidden");
  $("usuario").textContent = "Usuario: " + usuarioActual.email;
  await cargarMeses();
}

async function cerrarSesion(){
  await sb.auth.signOut();
  usuarioActual = null; meses = []; movimientos = [];
  $("app").classList.add("hidden");
  $("loginScreen").classList.remove("hidden");
  mostrarMensaje("","");
  limpiarTabla();
}

async function cargarMeses(){
  const {data,error} = await sb.from("meses").select("*")
    .eq("user_id",usuarioActual.id)
    .order("anio",{ascending:false}).order("mes",{ascending:false});
  if(error) return alert("Error cargando los meses: " + error.message);
  meses = data || [];
  const actual = $("mesSeleccionado").value;
  const sel = $("mesSeleccionado");
  sel.innerHTML = '<option value="">Selecciona un mes</option>';
  meses.forEach(m => {
    const o = document.createElement("option");
    o.value = m.id; o.textContent = m.nombre + " " + m.anio;
    sel.appendChild(o);
  });
  if(meses.some(m => String(m.id) === actual)) sel.value = actual;
  mostrarListaMeses();
}

function mostrarListaMeses(){
  const c = $("listaMeses");
  c.innerHTML = "";
  meses.forEach(m => {
    const div = document.createElement("div");
    div.className = "mes-item";
    const t = document.createElement("span");
    t.textContent = "📅 " + m.nombre + " " + m.anio;
    const b = document.createElement("button");
    b.textContent = "Eliminar"; b.className = "delete";
    b.onclick = () => eliminarMes(m.id);
    div.append(t,b);
    c.appendChild(div);
  });
}

const NUM_MES = {enero:1,febrero:2,marzo:3,abril:4,mayo:5,junio:6,julio:7,agosto:8,septiembre:9,setiembre:9,octubre:10,noviembre:11,diciembre:12};
const capitalizar = t => t.charAt(0).toUpperCase() + t.slice(1).toLowerCase();

async function crearMes(){
  const nombre = $("nombreMes").value.trim();
  const anio = parseInt($("anioMes").value);
  if(!nombre) return alert("Escribe el nombre del mes.");
  if(!anio || anio < 2000 || anio > 2100) return alert("Escribe un año válido.");
  const numero = NUM_MES[nombre.toLowerCase()];
  if(!numero) return alert("Escribe un mes válido. Ejemplo: Enero.");
  if(meses.some(m => m.mes === numero && m.anio === anio)) return alert("Este mes ya existe.");
  const {data,error} = await sb.from("meses")
    .insert({user_id:usuarioActual.id, nombre:capitalizar(nombre), mes:numero, anio})
    .select().single();
  if(error) return alert("Error creando el mes: " + error.message);
  $("nombreMes").value = ""; $("anioMes").value = "";
  await cargarMeses();
  $("mesSeleccionado").value = data.id;
  await cargarMes();
}

async function eliminarMes(id){
  if(!confirm("¿Seguro que quieres eliminar este mes y todos sus movimientos?")) return;
  const {error} = await sb.from("meses").delete().eq("id",id);
  if(error) return alert("Error eliminando el mes: " + error.message);
  if($("mesSeleccionado").value === String(id)){ $("mesSeleccionado").value = ""; movimientos = []; limpiarTabla(); }
  await cargarMeses();
}

async function cargarMes(){
  const mesId = $("mesSeleccionado").value;
  if(!mesId){ movimientos = []; return limpiarTabla(); }
  const {data,error} = await sb.from("movimientos").select("*")
    .eq("mes_id",mesId).eq("user_id",usuarioActual.id)
    .order("fecha",{ascending:false}).order("creado_en",{ascending:false});
  if(error) return alert("Error cargando movimientos: " + error.message);
  movimientos = data || [];
  mostrarMovimientos();
}

async function agregarMovimiento(){
  const mesId = $("mesSeleccionado").value;
  if(!mesId) return alert("Primero selecciona un mes.");
  const descripcion = $("descripcion").value.trim();
  const monto = parseFloat($("monto").value);
  const tipo = $("tipo").value;
  if(!descripcion) return alert("Escribe una descripción.");
  if(!monto || monto <= 0) return alert("Escribe un monto válido.");
  const {error} = await sb.from("movimientos")
    .insert({user_id:usuarioActual.id, mes_id:mesId, fecha:hoyLocal(), descripcion, tipo, monto});
  if(error) return alert("Error guardando movimiento: " + error.message);
  $("descripcion").value = ""; $("monto").value = "";
  await cargarMes();
}

function mostrarMovimientos(){
  const tabla = $("tablaMovimientos");
  tabla.innerHTML = "";
  if(movimientos.length === 0){
    $("sinMovimientos").textContent = "No hay movimientos en este mes.";
    $("sinMovimientos").style.display = "block";
    return actualizarTotales();
  }
  $("sinMovimientos").style.display = "none";
  movimientos.forEach(m => {
    const tr = document.createElement("tr");
    const celdas = [m.fecha, m.descripcion, m.tipo === "ingreso" ? "Ingreso" : "Egreso", dinero(m.monto)];
    celdas.forEach(txt => { const td = document.createElement("td"); td.textContent = txt; tr.appendChild(td); });
    const td = document.createElement("td");
    const b = document.createElement("button");
    b.textContent = "Eliminar"; b.className = "delete";
    b.onclick = () => eliminarMovimiento(m.id);
    td.appendChild(b); tr.appendChild(td);
    tabla.appendChild(tr);
  });
  actualizarTotales();
}

function actualizarTotales(){
  let ing = 0, egr = 0;
  movimientos.forEach(m => { if(m.tipo === "ingreso") ing += Number(m.monto); else egr += Number(m.monto); });
  $("totalIngresos").textContent = dinero(ing);
  $("totalEgresos").textContent = dinero(egr);
  $("saldo").textContent = dinero(ing - egr);
}

async function eliminarMovimiento(id){
  if(!confirm("¿Quieres eliminar este movimiento?")) return;
  const {error} = await sb.from("movimientos").delete().eq("id",id);
  if(error) return alert("Error eliminando el movimiento: " + error.message);
  await cargarMes();
}

function limpiarTabla(){
  $("tablaMovimientos").innerHTML = "";
  $("sinMovimientos").textContent = "Selecciona un mes para comenzar.";
  $("sinMovimientos").style.display = "block";
  ["totalIngresos","totalEgresos","saldo"].forEach(id => $(id).textContent = "Q 0.00");
}

async function comprobarSesion(){
  if(!sb) return;
  const {data} = await sb.auth.getSession();
  if(data.session){ usuarioActual = data.session.user; mostrarAplicacion(); }
}
comprobarSesion();
